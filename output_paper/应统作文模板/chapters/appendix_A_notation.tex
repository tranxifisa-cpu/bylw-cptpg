\subsection{Dirichlet得分矩}

\begin{lemma}[增广轨迹得分的统一矩界]\label{lem:dirichlet-moments}
在假设~\ref{ass:trajectory}--\ref{ass:likelihood}下，存在$0<\underline a\le\overline a<\infty$，使对所有$i,u,\theta$均有
\begin{equation}\label{eq:concentration-compactness}
 \underline a\le a_{i,u}(\theta)\le\overline a.
\end{equation}
对每个固定的$1\le p<\infty$，
\begin{equation}\label{eq:score-moment-bound}
 \sup_{k,\theta}\E\norm{G_{k,\theta}}_2^p<\infty.
\end{equation}
轨迹对数似然的Hessian矩阵$\mathcal H_{k,\theta}$亦满足
\begin{equation}\label{eq:hessian-moment-bound}
 \sup_{k,\theta}\E\norm{\mathcal H_{k,\theta}}_{\op}<\infty.
\end{equation}
进一步，对任意有界可测的轨迹泛函$F$，条件轨迹分布下可交换至二阶求导与积分：
\begin{equation}\label{eq:likelihood-ratio-identities}
 \nabla_\theta\E_\theta F=\E_\theta(FG_{k,\theta}),
 \qquad
 \nabla_\theta^2\E_\theta F
 =\E_\theta\{F(G_{k,\theta}G_{k,\theta}^\top+\mathcal H_{k,\theta})\}.
\end{equation}
\end{lemma}

\begin{proof}
$\Theta$的紧性与$q_{i,u}$的有界性给出式~\eqref{eq:concentration-compactness}。Dirichlet分布的每一坐标都具有Beta边缘分布，其两个参数落在$(0,\infty)^2$的紧子集中。因此对任意有限的$p\ge1$，
\[
 \sup_{\mathfrak a,\mathfrak b}\int_0^1
 \abs{\log x}^p x^{\mathfrak a-1}(1-x)^{\mathfrak b-1}\,dx<\infty,
\]
其中上确界取遍诱导的紧参数集。digamma与polygamma项在相同的正紧区间上有界。结合式~\eqref{eq:trajectory-score}、因子有界性和有限的$(h,N)$，得到式~\eqref{eq:score-moment-bound}；再对得分求一次导数，得到式~\eqref{eq:hessian-moment-bound}。依假设~\ref{ass:likelihood}，市场转移核与$\mathsf M_u(\cdot\mid s_u)$不贡献似然导数。

为交换求导与轨迹积分，在仍包含于策略定义域的前提下将$\Theta$略微扩张。此邻域中的浓度参数保持在某个$[a_-,a_+]$内，其中$a_->0$。对单次潜在动作$\widetilde\omega$，Dirichlet密度及其关于$\theta$的一、二阶导数，都由与$\theta$无关的常数倍下式控制：
\[
 \prod_{i=0}^{N}\widetilde\omega_i^{a_- -1}
 \left(1+\sum_{i=0}^{N}|\log\widetilde\omega_i|^2\right).
\]
当$a_->0$时，这一包络在单纯形上可积。有限的$h$次潜在动作相乘后仍构成可积的轨迹包络；与参数无关的市场核在每步条件积分中均积分为一。因此，有界$F$与前两阶导数的乘积均被同一可积函数支配。应用控制收敛定理得到式~\eqref{eq:likelihood-ratio-identities}。
\end{proof}

\subsection{CPT目标的光滑性}

\begin{lemma}[统一梯度界与光滑界]\label{lem:cpt-smoothness}
在假设~\ref{ass:trajectory}--\ref{ass:cpt-regularity}下，存在$B_g,L<\infty$，使
\begin{equation}\label{eq:uniform-gradient-smoothness}
 \sup_{k,\theta\in\Theta}\norm{g_k(\theta)}_2\le B_g,
 \qquad
 \norm{g_k(\theta)-g_k(\theta')}_2\le L\norm{\theta-\theta'}_2.
\end{equation}
记
\[
 C_{G,2}:=\sup_{k,\theta}\E\norm{G_{k,\theta}}_2^2,
 \qquad
 C_H:=\sup_{k,\theta}\E\norm{\mathcal H_{k,\theta}}_{\op},
\]
则可取
\begin{equation}\label{eq:smoothness-constant}
 B_g=2B_vC_1\sqrt{C_{G,2}},
 \qquad
 L=2B_v\{C_2C_{G,2}+C_1(C_{G,2}+C_H)\}.
\end{equation}
此外，$\abs{J_k(\theta,r_k)}\le B_v$。
\end{lemma}

\begin{proof}
对侧别$\sigma$与阈值$y$，记$I_\sigma(y):=\1\{v_\sigma(Y)>y\}$。由似然得分恒等式及引理~\ref{lem:dirichlet-moments}，
\[
 \nabla S_\theta^\sigma(y)=\E\{I_\sigma(y)G_\theta\},
 \qquad
 \nabla^2S_\theta^\sigma(y)
 =\E\{I_\sigma(y)(G_\theta G_\theta^\top+\mathcal H_\theta)\}.
\]
指标函数有界，得分矩也有界，因此可由控制收敛交换求导与期望，并有
\[
 \norm{\nabla S_\theta^\sigma(y)}_2\le\sqrt{C_{G,2}},
 \qquad
 \norm{\nabla^2S_\theta^\sigma(y)}_{\op}\le C_{G,2}+C_H.
\]
对CPT积分求两次导数，再用假设~\ref{ass:cpt-regularity}，得到
\[
 \nabla^2\!\int_0^{B_v}w_\sigma^\eps(S_\theta^\sigma(y))dy
 =\int_0^{B_v}
 \{(w_\sigma^\eps)''(S)\nabla S\nabla S^\top
 +(w_\sigma^\eps)'(S)\nabla^2S\}dy.
\]
被积函数由上述常数统一控制，故有式~\eqref{eq:smoothness-constant}；一阶导数的界同理。最后，$0\le w_\sigma^\eps\le1$且两个积分的共同区间长度为$B_v$，所以$\abs{J_k}\le B_v$。
\end{proof}

\subsection{投影残差与受约束驻点}

\begin{lemma}[投影残差刻画一阶驻点]\label{lem:residual-stationarity}
设$\Theta$闭且凸，$\theta\in\Theta$，$\gamma,\vartheta>0$，$g\in\R^d$，则
\begin{equation}\label{eq:projection-stationarity-equivalence}
 \Pi_\Theta\{\theta+\gamma\mathsf d(g)\}=\theta
 \quad\Longleftrightarrow\quad
 \ip{g}{\theta'-\theta}\le0\quad\forall \theta'\in\Theta.
\end{equation}
因此，式~\eqref{eq:stationary-set}中的两种驻点定义等价。
\end{lemma}

\begin{proof}
投影的变分不等式表明，$x=\Pi_\Theta(y)$当且仅当对所有$\theta'\in\Theta$有$\ip{y-x}{\theta'-x}\le0$。代入$y=\theta+\gamma\mathsf d(g)$，并注意$\mathsf d(g)=g/\max\{\norm g_2,\vartheta\}$是$g$的正倍数，即得结论。
\end{proof}
