\chapter{算法设计}\label{sec:algorithm}

每一期的学习从已实现的市场、财富、参考点和持仓出发。在这个共同起点下，模拟模型产生多条候选投资路径，用于估计当期CPT目标的梯度；真实账户则沿一条独立的随机策略路径前进，逐日更新财富和参考点。两条路径使用相同的策略分布和交易规则，而参数只在时期结束后更新。

\section{中心化留一法与随机层修正}

固定一期的起点信息、模型和参数，暂时省略时期下标。一条完整轨迹可概括为
\begin{equation}\label{eq:trajectory-summary}
 \mathsf X_i=(V_i^+,V_i^-,G_i),
 \qquad
 V_i^\pm=v_\pm(Y_i).
\end{equation}
其中$V_i^+$和$V_i^-$分别是该轨迹的收益侧与损失侧价值，$G_i$是整条轨迹的得分。对$\sigma\in\{+,-\}$，令$\varsigma_+=1$、$\varsigma_-=-1$，并定义留一生存概率
\begin{equation}\label{eq:loo-survival}
 I_{i,\sigma}(y)=\1\{V_i^\sigma>y\},
 \qquad
 \widehat S_{-i,n}^\sigma(y)
 =\frac1{n-1}\sum_{j\ne i}I_{j,\sigma}(y).
\end{equation}
第$i$条轨迹的生存事件与用于评价其分位位置的其余$n-1$条轨迹分开，避免自身参与自身的概率估计。中心化留一统计量为
\begin{equation}\label{eq:centered-loo}
 U_n^{\rm cen}
 =\frac1n\sum_{i=1}^{n}G_i
 \sum_{\sigma\in\{+,-\}}\varsigma_\sigma
 \int_0^{B_v}
 (w_\sigma^\eps)'\!\left(\widehat S_{-i,n}^\sigma(y)\right)
 \left\{I_{i,\sigma}(y)-\widehat S_{-i,n}^\sigma(y)\right\}dy.
\end{equation}
供层间差分使用的未中心化统计量为
\begin{equation}\label{eq:raw-loo}
 U_n^{\rm raw}
 =\frac1n\sum_{i=1}^{n}G_i
 \sum_{\sigma\in\{+,-\}}\varsigma_\sigma
 \int_0^{B_v}
 (w_\sigma^\eps)'\!\left(\widehat S_{-i,n}^\sigma(y)\right)
 I_{i,\sigma}(y)\,dy.
\end{equation}
两者期望相同，但随机波动的结构不同。中心化统计量用作低方差基项，未中心化统计量用于构造严格抵消一阶样本影响的全样本与半样本差分。

取$n\ge2$、激活概率$p_{\rm act}\in(0,1]$和指数$b\in(1,2)$，令
\begin{equation}\label{eq:level-probabilities}
 \nu_\ell=(1-2^{-b})2^{-b(\ell-1)},
 \qquad \ell=1,2,\ldots.
\end{equation}
在层级$\ell$上，$n_\ell=n2^\ell$条轨迹同时构成一个全样本和两个互不相交的半样本。相应差分为
\begin{equation}\label{eq:raw-level-difference}
 D_{n,\ell}^{\rm raw}
 =U_{n_\ell}^{\rm raw}
 -\frac12\left(
 U_{n_{\ell-1}}^{\rm raw,(1)}
 +U_{n_{\ell-1}}^{\rm raw,(2)}
 \right).
\end{equation}
设$\mathsf A\sim\operatorname{Bernoulli}(p_{\rm act})$；在$\mathsf A=1$时，从分布$\nu$抽取$\ell^\star$。激活变量和层级均与基项样本及修正样本独立。一次完整的去偏输出为
\begin{equation}\label{eq:debiased-output}
 Z_n^{\db}
 =U_n^{\rm cen}
 +\frac{\mathsf A}{p_{\rm act}\nu_{\ell^\star}}D_{n,\ell^\star}^{\rm raw},
\end{equation}
其中$\mathsf A=0$时第二项定义为零。

\begin{algorithm}[H]
\caption{一次中心化留一随机层去偏CPT梯度输出}\label{alg:cloo-rd}
\begin{minipage}{0.88\linewidth}
\begin{algorithmic}[1]
\REQUIRE $(z,r,\widetilde P,\theta)$，$n\ge2$，$p_{\rm act}\in(0,1]$，$b\in(1,2)$
\STATE 在冻结的$(z,r,\theta)$下独立抽取$\mathcal X_0\gets\{\mathsf X_i\}_{i=1}^{n}$；计算$U_0\gets U_n^{\rm cen}(\mathcal X_0)$
\STATE 抽取$\mathsf A\sim\operatorname{Bernoulli}(p_{\rm act})$
\IF{$\mathsf A=0$}
    \RETURN $U_0$
\ENDIF
\STATE 抽取$\ell\sim\nu$；从同一冻结模型独立抽取$\mathcal X_\ell\gets\{\mathsf X_i\}_{i=1}^{n2^\ell}$，且与$\mathcal X_0$独立
\STATE $D\gets U_{n2^\ell}^{\rm raw}(\mathcal X_\ell)-\tfrac12\{U_{n2^{\ell-1}}^{\rm raw}(\mathcal X_\ell^{(1)})+U_{n2^{\ell-1}}^{\rm raw}(\mathcal X_\ell^{(2)})\}$
\RETURN $Z_n^{\db}=U_0+D/(p_{\rm act}\nu_\ell)$
\end{algorithmic}
\end{minipage}
\end{algorithm}

以一期五个交易日为例，先从当期账户状态模拟$n$条五日投资路径，形成中心化留一基项；若随机修正被激活，再独立生成抽中层级所需的路径。全样本与两个半样本复用这批新路径，使层间差分反映样本精度变化，而不是不同市场噪声造成的波动。

\section{在线参数更新}

第$k$期从冻结模拟模型独立取得$M_k$次式~\eqref{eq:debiased-output}的输出，并求平均：
\begin{equation}\label{eq:gradient-average}
 \widehat g_k^{\db}
 =\frac1{M_k}\sum_{m=1}^{M_k}Z_{k,m}^{\db}.
\end{equation}
时期结束后，按归一化方向更新参数：
\begin{equation}\label{eq:parameter-update}
 \theta_{k+1}
 =\Pi_\Theta\left\{\theta_k+\gamma\mathsf d(\widehat g_k^{\db})\right\}.
\end{equation}

\begin{algorithm}[H]
\caption{动态参考点CPT策略梯度（\DRCPTPG）}\label{alg:drcptpg}
\begin{minipage}{0.88\linewidth}
\begin{algorithmic}[1]
\REQUIRE $\theta_1\in\Theta$，$(X_1,r_1,\omega_0)$，$h,\chi,\gamma,\vartheta$，$(n,p_{\rm act},b)$，$\{M_k\}_{k=1}^{K}$
\FOR{$k=1,\ldots,K$}
    \STATE 根据$\cF_k$固定$(z_k,r_k,\widetilde P_k)$
    \STATE 由算法~\ref{alg:cloo-rd}独立生成$Z_{k,1}^{\db},\ldots,Z_{k,M_k}^{\db}$
    \STATE $\widehat g_k^{\db}\gets M_k^{-1}\sum_{m=1}^{M_k}Z_{k,m}^{\db}$
    \FOR{$u=t_k,\ldots,t_k+h-1$}
        \STATE $\widetilde\omega_u\sim\pi_{\theta_k}(\cdot\mid s_u)$
        \STATE $\omega_u\gets(1-\chi)\omega_{u-1}+\chi\mathsf M_u(\widetilde\omega_u\mid s_u)$
        \STATE $X_{u+1}\gets X_u\{1+\omega_u^\top R_{u+1}-c_{\rm tc}\norm{\omega_u-\omega_{u-1}}_1\}$；$r_{u+1}\gets\Phi(r_u,X_{u+1})$
    \ENDFOR
    \STATE $\theta_{k+1}\gets\Pi_\Theta\{\theta_k+\gamma\mathsf d(\widehat g_k^{\db})\}$
\ENDFOR
\RETURN $\{\theta_k,X_{t_k},r_k,\omega_{t_k-1}\}_{k=1}^{K+1}$
\end{algorithmic}
\end{minipage}
\end{algorithm}

仍以五日调仓为例，期初已知的财富、参考点与持仓成为所有模拟轨迹的共同起点。模拟轨迹各自随机配置股票和现金，逐日经历收益、成本及参考点变化，最终汇总成当期梯度估计。实际账户也按同一随机Dirichlet策略执行，每天记录新财富和新参考点。五日结束后，算法才依据模拟轨迹提供的梯度方向更新因子参数，并将真实账户的期末状态带入下一期。期内的市场变化由各条轨迹完整承受，参数则始终保持为期初确定的值。
