\chapter{问题设定与假设}\label{sec:setup}

固定参考点的投资模型中，一次调仓后的财富变化只改变账户余额；这里，财富变化还会改变下一次判断盈亏的基准。市场收益、账户财富和参考点沿同一条交易路径相互作用，使策略每期面对的CPT目标随之变化。下文从日常调仓和参考点递推入手，定义条件目标与随机策略，最后以投影残差衡量跨期跟踪。

\section{非平稳有限时域投资过程}

将投资过程划分为$K$个时期，记为$k=1,\ldots,K$。每期从$t_k$开始，包含固定的$h\ge1$个决策日，相邻时期满足$t_{k+1}=t_k+h$。在日历时点$u\in\{t_k,\ldots,t_k+h-1\}$，状态与组合权重分别为
\begin{equation}\label{eq:state-action}
 s_u=(m_u,X_u,r_u,\omega_{u-1}),
 \qquad
 \omega_u\in\Delta_N:=\left\{\omega\in\R_+^{N+1}:\sum_{i=0}^{N}\omega_i=1\right\},
\end{equation}
其中$m_u$是已观测的市场状态，资产$0$为现金，$X_u>0$为以初始账户价值归一化的财富，$r_u\ge0$为同一单位下的参考财富。设收益向量为$R_{u+1}$，现金收益$R_{0,u+1}=0$，比例交易成本系数为$c_{\rm tc}\ge0$，则
\begin{equation}\label{eq:wealth-update}
 X_{u+1}=X_u\left(1+\omega_u^\top R_{u+1}
 -c_{\rm tc}\norm{\omega_u-\omega_{u-1}}_1\right),
 \qquad X_1=1.
\end{equation}
给定当前状态与执行动作后的市场转移规律可以随日历时间变化。因此，非平稳性既可能发生在不同时期之间，也可能出现在一个时期内部。策略梯度所需的参数依赖结构由后文假设~\ref{ass:likelihood}规定，并不要求时期内部市场平稳。

\section{动态参考点与终端相对结果}

令$\etaP,\etaM\in[0,1)$，参考点按照当日财富相对于此前参考点的位置更新：
\begin{equation}\label{eq:reference-update}
 r_{u+1}=\Phi(r_u,X_{u+1})
 =\begin{cases}
 (1-\etaP)r_u+\etaP X_{u+1}, & X_{u+1}\ge r_u,\\[1mm]
 (1-\etaM)r_u+\etaM X_{u+1}, & X_{u+1}<r_u.
 \end{cases}
\end{equation}
两种适应速度可以不同。例如，若财富上升后的参考点上调较快，先涨后跌的路径即使最终回到同一财富水平，也可能被评价为损失。设$\cF_k$表示第$k$期开始时可用的信息，并记
\begin{equation}\label{eq:start-context}
 r_k:=r_{t_k},
 \qquad
 z_k:=(m_{t_k},X_{t_k},\omega_{t_k-1}).
\end{equation}
这里$r_k$专指时期起点的参考点，$r_u$则表示时期内部随日更新的参考点。第$k$期的蒙特卡洛学习轨迹均以冻结的$(\cF_k,z_k,r_k)$为共同起点，但每条轨迹沿自身的财富路径递推参考点。轨迹$\tau$的终端相对结果为
\begin{equation}\label{eq:relative-outcome}
 Y_k(\tau)=X_{t_k+h}(\tau)-r_{t_k+h}(\tau).
\end{equation}
由此，同一时期的多条候选轨迹可以落入不同的盈利或亏损区域。

\section{Dirichlet投资组合策略}

资产$i=0,\ldots,N$在时点$u$的因子特征为$q_{i,u}\in\R^d$，由当前市场信息和状态$s_u$中的前期持仓构造。给定参数$\theta\in\Theta\subset\R^d$，策略先随机产生潜在组合：
\begin{equation}\label{eq:dirichlet-policy}
 a_{i,u}(\theta)=\exp(\theta^\top q_{i,u}),
 \qquad
 \widetilde\omega_u\sim
 \pi_\theta(\cdot\mid s_u)
 =\Dir\!\left(a_{0,u}(\theta),\ldots,a_{N,u}(\theta)\right).
\end{equation}
参数$\theta$决定各因子的配置作用，因子值因股票而异，因此同一参数可以形成不同的资产浓度。设$\mathsf M_u(\cdot\mid s_u):\Delta_N\to\Delta_N$为依赖已观测历史的可行性映射，$\chi\in(0,1]$为固定交易比例。实际执行的组合为
\begin{equation}\label{eq:action-map}
 \omega_u=(1-\chi)\omega_{u-1}+\chi\,\mathsf M_u(\widetilde\omega_u\mid s_u).
\end{equation}
真实账户路径和学习轨迹均按式~\eqref{eq:dirichlet-policy}随机抽取组合，再经式~\eqref{eq:action-map}执行。增广轨迹保留潜在组合$\widetilde\omega_u$，其得分函数为
\begin{equation}\label{eq:trajectory-score}
 G_{k,\theta}(\tau)
 =\sum_{u=t_k}^{t_k+h-1}\sum_{i=0}^{N}
 a_{i,u}q_{i,u}
 \left\{\log\widetilde\omega_{i,u}
 -\psi(a_{i,u})+\psi\!\left(\sum_{j=0}^{N}a_{j,u}\right)\right\},
\end{equation}
其中$\psi$是digamma函数。得分记录了整条轨迹对策略参数的似然敏感性，后文将其与轨迹在CPT分布中的相对位置结合，得到策略梯度。

\section{正则化CPT目标}

对$x\ge0$定义
\begin{equation}\label{eq:value-functions}
 u_{\alpha,\eps_v}(x)=(x^2+\eps_v^2)^{\alpha/2}-\eps_v^\alpha,
 \qquad
 v_+(y)=u_{\alpha_+,\eps_v}(y^+),
 \qquad
 v_-(y)=\lambda u_{\alpha_-,\eps_v}(y^-),
\end{equation}
其中$y^+:=\max\{y,0\}$、$y^-:=\max\{-y,0\}$，且$0<\alpha_+,\alpha_-\le1$、$\lambda>1$、$\eps_v>0$。正则化的价值函数在零点附近保持平滑，同时保留收益与损失的不对称评价。设$w_\pm$为严格递增的概率权重函数。取固定的$\eps\in(0,1/2)$，其端点归一化形式为
\begin{equation}\label{eq:normalized-weight}
 w_\pm^\eps(p)
 =\frac{w_\pm(\eps+(1-2\eps)p)-w_\pm(\eps)}
 {w_\pm(1-\eps)-w_\pm(\eps)},
 \qquad 0\le p\le1.
\end{equation}
于是$w_\pm^\eps(0)=0$且$w_\pm^\eps(1)=1$。

固定一期的环境或模型$P$及起点$(z,r)$后，$P_\theta$表示策略$\pi_\theta$诱导的增广轨迹分布。记
\[
 S_{P,z,r,\theta}^{\pm}(y)
 =P_\theta\{v_\pm(Y)>y\}.
\]
在假设~\ref{ass:trajectory}规定的分析域内，$v_\pm(Y)\in[0,B_v]$。条件CPT目标定义为
\begin{equation}\label{eq:generic-cpt-objective}
 J(P,z,r;\theta)
 =\int_0^{B_v}w_+^\eps\!\left(S_{P,z,r,\theta}^{+}(y)\right)dy
 -\int_0^{B_v}w_-^\eps\!\left(S_{P,z,r,\theta}^{-}(y)\right)dy.
\end{equation}
设$P_k$为第$k$期真实的条件模型，$\widetilde P_k$为该期冻结的模拟模型。求导前固定起点信息和模型；参数如何进入轨迹分布由假设~\ref{ass:likelihood}给出。真实目标与模拟目标分别为
\begin{equation}\label{eq:true-sim-objectives}
 J_k(\theta,r_k):=J(P_k,z_k,r_k;\theta),
 \qquad
 \widetilde J_k(\theta,r_k):=J(\widetilde P_k,z_k,r_k;\theta),
\end{equation}
对应梯度为
\begin{equation}\label{eq:true-sim-gradients}
 g_k(\theta):=\nabla_\theta J_k(\theta,r_k),
 \qquad
 \widetilde g_k(\theta):=\nabla_\theta\widetilde J_k(\theta,r_k).
\end{equation}

\section{在线近似误差与目标变化}

在实际迭代点定义模拟梯度与真实梯度的差异
\begin{equation}\label{eq:simulator-gradient-error}
 \beta_k
 :=\norm{\widetilde g_k(\theta_k)-g_k(\theta_k)}_2,
\end{equation}
以及真实目标序列的累计变化
\begin{equation}\label{eq:objective-variation}
 \cV_K
 :=\sum_{k=1}^{K-1}
 \sup_{\theta\in\Theta}
 \abs{J_{k+1}(\theta,r_{k+1})-J_k(\theta,r_k)}.
\end{equation}
$\beta_k$衡量冻结模拟模型的梯度近似误差，$\cV_K$则衡量真实目标随时间移动的程度。两者来源不同，在在线界中也分别出现。下述有界条件保证它们在任意有限期内有定义。

\section{投影残差与动态局部遗憾}

给定归一化下界$\vartheta>0$与步长$\gamma>0$，定义方向映射
\begin{equation}\label{eq:normalized-direction}
 \mathsf d(g)=\frac{g}{\max\{\norm{g}_2,\vartheta\}},
\end{equation}
以及真实梯度对应的投影残差
\begin{equation}\label{eq:projected-residual}
 Q_k(\theta)
 =\frac{\Pi_\Theta\{\theta+\gamma\mathsf d(g_k(\theta))\}-\theta}{\gamma}.
\end{equation}
其中$\Pi_\Theta$表示到$\Theta$的欧氏投影。受约束的一阶驻点集合为
\begin{equation}\label{eq:stationary-set}
 \Sta_k
 =\{\theta\in\Theta:Q_k(\theta)=0\}
 =\left\{\theta\in\Theta:
 \ip{g_k(\theta)}{\theta'-\theta}\le0\ \forall \theta'\in\Theta\right\}.
\end{equation}
由于$J_k$在紧凸集$\Theta$上连续可微，每期目标都取得最大值，且最大值满足式~\eqref{eq:stationary-set}的一阶条件；$Q_k$的连续性进一步保证$\Sta_k$非空且紧。沿实际迭代路径，记
\begin{equation}\label{eq:periodwise-residual}
 \zeta_k:=\norm{Q_k(\theta_k)}_2^2,
 \qquad
 \cE_K:=\frac1K\sum_{k=1}^{K}\E\zeta_k.
\end{equation}
给定窗口长度$W\ge1$与折扣系数$\rho\in(0,1]$，定义
\begin{equation}\label{eq:dynamic-window-residual}
 A_{\rho,W}:=\sum_{j=0}^{W-1}\rho^j,
 \qquad
 \bar Q_k
 :=A_{\rho,W}^{-1}\sum_{j=0}^{W-1}\rho^j Q_{k-j}(\theta_{k-j}),
\end{equation}
并约定$t\le0$时$Q_t(\theta_t)=0$。动态局部遗憾为
\begin{equation}\label{eq:dynamic-regret}
 \Reg_{\rho,W}(K):=\sum_{k=1}^{K}\norm{\bar Q_k}_2^2.
\end{equation}
$\zeta_k$记录当期一阶残差；动态局部遗憾则先对最近各期的残差向量作时间平滑，再对平滑向量的平方范数逐期累加。这里不是对各期平方残差先求平均。

\section{分析假设}\label{sec:assumptions}

以下条件分别约束轨迹的分析域、轨迹似然中的参数来源和CPT权重的光滑性。它们使固定目标的梯度估计与跨期的一阶跟踪使用同一组基本量。

\begin{assumption}[策略参数与轨迹的共同有界域]\label{ass:trajectory}
参数集合$\Theta\subset\R^d$非空、紧且凸，并包含于策略有定义的一个开集中。时期长度$h$和资产数$N$有限。对所有分析中的真实模型$P_k$与冻结模拟模型$\widetilde P_k$，统一有
\begin{equation}\label{eq:primitive-bounds}
 \sup_{i,u}\norm{q_{i,u}}_2<\infty,
 \qquad
 \sup_u X_u<\infty,
 \qquad
 \sup_u r_u<\infty,
 \qquad
 0\le v_\pm(Y_k)\le B_v<\infty.
\end{equation}
\end{assumption}

参数域、时期长度、资产数量和因子范数由模型设计控制；财富、参考点与价值函数的上界则指定理论作用的共同分析域。该条件保证不同期的得分矩、目标值及光滑常数可以统一控制。

\begin{assumption}[由策略传递的似然参数依赖]\label{ass:likelihood}
给定已观测历史与执行动作，市场转移核和依赖历史的可行性映射$\mathsf M_u(\cdot\mid s_u)$不显式依赖$\theta$。潜在Dirichlet提议$\widetilde\omega_u$保留在增广轨迹中；当$\theta$变化时，每期的冻结模拟模型保持不变。因此，条件增广轨迹分布对$\theta$的显式依赖仅来自策略密度。
\end{assumption}

在这一结构下，市场及交易可行性机制可以通过执行动作和历史受到策略影响，却不产生额外的显式似然导数，因而整条轨迹的参数敏感性由式~\eqref{eq:trajectory-score}给出。若转移核本身直接含有策略参数，则梯度恒等式还须包含相应的似然导数或路径导数。

\begin{assumption}[正则化CPT权重的光滑性]\label{ass:cpt-regularity}
式~\eqref{eq:value-functions}--\eqref{eq:normalized-weight}中的偏好参数固定。归一化权重函数$w_\pm^\eps$严格递增，属于$C^3([0,1])$，并满足
\begin{equation}\label{eq:weight-derivative-bounds}
 C_j
 :=\max_{\sigma\in\{+,-\}}\sup_{p\in[0,1]}
 \abs{(w_\sigma^\eps)^{(j)}(p)}
 <\infty,
 \qquad j=1,2,3.
\end{equation}
\end{assumption}

一阶导数用于CPT得分梯度恒等式，二阶导数控制目标的光滑性，三阶导数则控制全样本与两个半样本耦合后的余项。附录~\ref{app:notation}由此建立统一的有限得分矩，以及对真实目标和冻结模拟目标均适用的$B_g,L<\infty$，使
\begin{equation}\label{eq:gradient-smoothness-summary}
 \sup_{k,\theta\in\Theta}\norm{g_k(\theta)}_2\le B_g,
 \qquad
 \norm{g_k(\theta)-g_k(\theta')}_2
 \le L\norm{\theta-\theta'}_2.
\end{equation}
特别地，任意有限实验中的$\beta_k$和$\cV_K$都是有限的。
