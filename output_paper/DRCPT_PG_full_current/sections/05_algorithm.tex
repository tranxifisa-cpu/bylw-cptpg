\section{Algorithm}\label{sec:algorithm}

\subsection{Centered leave-one-out base and raw randomized correction}

Fix one frozen episode context and parameter, and suppress the episode index.  A complete trajectory yields
\begin{equation}\label{eq:trajectory-summary}
 \mathsf X_i=(V_i^+,V_i^-,G_i),
 \qquad
 V_i^\pm=v_\pm(Y_i).
\end{equation}
For $\sigma\in\{+,-\}$, let $\varsigma_+=1$, $\varsigma_-=-1$ and define
\begin{equation}\label{eq:loo-survival}
 I_{i,\sigma}(y)=\1\{V_i^\sigma>y\},
 \qquad
 \widehat S_{-i,n}^\sigma(y)
 =\frac1{n-1}\sum_{j\ne i}I_{j,\sigma}(y).
\end{equation}
The centered leave-one-out statistic is
\begin{equation}\label{eq:centered-loo}
 U_n^{\rm cen}
 =\frac1n\sum_{i=1}^{n}G_i
 \sum_{\sigma\in\{+,-\}}\varsigma_\sigma
 \int_0^{B_v}
 (w_\sigma^\eps)'\!\left(\widehat S_{-i,n}^\sigma(y)\right)
 \left\{I_{i,\sigma}(y)-\widehat S_{-i,n}^\sigma(y)\right\}dy.
\end{equation}
The corresponding raw statistic is
\begin{equation}\label{eq:raw-loo}
 U_n^{\rm raw}
 =\frac1n\sum_{i=1}^{n}G_i
 \sum_{\sigma\in\{+,-\}}\varsigma_\sigma
 \int_0^{B_v}
 (w_\sigma^\eps)'\!\left(\widehat S_{-i,n}^\sigma(y)\right)
 I_{i,\sigma}(y)\,dy.
\end{equation}

Choose $n\ge2$, activation probability $\varrho\in(0,1]$, and exponent $b\in(1,2)$.  Let
\begin{equation}\label{eq:level-probabilities}
 \nu_\ell=(1-2^{-b})2^{-b(\ell-1)},
 \qquad \ell=1,2,\ldots.
\end{equation}
For $n_\ell=n2^\ell$, compute $U_{n_\ell}^{\rm raw}$ on one batch of $n_\ell$ trajectories and the two half-sample statistics on its two disjoint halves.  Define
\begin{equation}\label{eq:raw-level-difference}
 D_{n,\ell}^{\rm raw}
 =U_{n_\ell}^{\rm raw}
 -\frac12\left(
 U_{n_{\ell-1}}^{\rm raw,(1)}
 +U_{n_{\ell-1}}^{\rm raw,(2)}
 \right).
\end{equation}
Let $\mathsf A\sim\operatorname{Bernoulli}(\varrho)$ and, conditional on $\mathsf A=1$, let $\ell^\star\sim\nu$; both are independent of the base sample and the correction batch.  One complete debiased output is
\begin{equation}\label{eq:debiased-output}
 Z_n^{\db}
 =U_n^{\rm cen}
 +\frac{\mathsf A}{\varrho\nu_{\ell^\star}}D_{n,\ell^\star}^{\rm raw},
\end{equation}
with the second term defined as zero when $\mathsf A=0$.

\begin{algorithm}[t]
\caption{One centered-LOO debiased CPT-gradient output}\label{alg:cloo-rd}
\begin{algorithmic}[1]
\Require $(z,r,\widetilde P,\theta)$, $n\ge2$, $\varrho\in(0,1]$, $b\in(1,2)$
\State $\mathcal X_0\gets\{\mathsf X_i\}_{i=1}^{n}$ i.i.d. from the frozen law at $(z,r,\theta)$; \quad $U_0\gets U_n^{\rm cen}(\mathcal X_0)$
\State $\mathsf A\sim\operatorname{Bernoulli}(\varrho)$
\If{$\mathsf A=0$}
    \State \Return $U_0$
\EndIf
\State $\ell\sim\nu$; \quad $\mathcal X_\ell\gets\{\mathsf X_i\}_{i=1}^{n2^\ell}$ i.i.d. from the same frozen law, independently of $\mathcal X_0$
\State $D\gets U_{n2^\ell}^{\rm raw}(\mathcal X_\ell)-\tfrac12\{U_{n2^{\ell-1}}^{\rm raw}(\mathcal X_\ell^{(1)})+U_{n2^{\ell-1}}^{\rm raw}(\mathcal X_\ell^{(2)})\}$
\State \Return $Z_n^{\db}=U_0+D/(\varrho\nu_\ell)$
\end{algorithmic}
\end{algorithm}

The level distribution is untruncated: no deterministic cap is imposed on $\ell^\star$, and completed outputs are not discarded as a function of the realized level or magnitude.

\subsection{Online update}

In episode $k$, generate $M_k$ conditionally independent copies of \eqref{eq:debiased-output} under the frozen simulator and set
\begin{equation}\label{eq:gradient-average}
 \widehat g_k^{\db}
 =\frac1{M_k}\sum_{m=1}^{M_k}Z_{k,m}^{\db}.
\end{equation}
The parameter update is
\begin{equation}\label{eq:parameter-update}
 \theta_{k+1}
 =\Pi_\Theta\left\{\theta_k+\gamma\mathsf d(\widehat g_k^{\db})\right\}.
\end{equation}

\begin{algorithm}[t]
\caption{Dynamic-reference CPT policy gradient (\DRCPTPG)}\label{alg:drcptpg}
\begin{algorithmic}[1]
\Require $\theta_1\in\Theta$, $(X_1,r_1,\omega_0)$, $h,\chi,\gamma,\vartheta$, $(n,\varrho,b)$, $\{M_k\}_{k=1}^{K}$
\For{$k=1,\ldots,K$}
    \State fix $(z_k,r_k,\widetilde P_k)$ conditionally on $\cF_k$
    \State generate $Z_{k,1}^{\db},\ldots,Z_{k,M_k}^{\db}$ independently by Algorithm~\ref{alg:cloo-rd}
    \State $\widehat g_k^{\db}\gets M_k^{-1}\sum_{m=1}^{M_k}Z_{k,m}^{\db}$
    \For{$u=t_k,\ldots,t_k+h-1$}
        \State $\widetilde\omega_u\sim\pi_{\theta_k}(\cdot\mid s_u)$; \quad $\omega_u\gets(1-\chi)\omega_{u-1}+\chi\mathsf M_u(\widetilde\omega_u)$
        \State $X_{u+1}\gets X_u\{1+\omega_u^\top R_{u+1}-c_{\rm tc}\norm{\omega_u-\omega_{u-1}}_1\}$; \quad $r_{u+1}\gets\Phi(r_u,X_{u+1})$
    \EndFor
    \State $\theta_{k+1}\gets\Pi_\Theta\{\theta_k+\gamma\mathsf d(\widehat g_k^{\db})\}$
\EndFor
\State \Return $\{\theta_k,X_{t_k},r_k,\omega_{t_k-1}\}_{k=1}^{K+1}$
\end{algorithmic}
\end{algorithm}
