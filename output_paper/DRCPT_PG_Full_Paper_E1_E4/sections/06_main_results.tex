\section{Main Results}\label{sec:results}

The theory follows the same chain as the method: endogenous reference updating first changes the objective being optimized; the frozen-target gradient must then be estimated without finite-budget plug-in bias; finally, the resulting updates must track a moving first-order target.  We state these claim-level results here and defer proof machinery---score moments, full/half cancellation, projected-ascent inequalities, and stationary-set drift calculations---to the appendices.

\subsection{Reference dependence}

\begin{proposition}[Same terminal wealth, different CPT value]\label{prop:path-dependence}
Let $x_{\rm lo}<r_0<x_{\rm f}<x_{\rm hi}$ and consider
\[
 \mathrm{I}: r_0\to x_{\rm hi}\to x_{\rm f},
 \qquad
 \mathrm{II}: r_0\to x_{\rm lo}\to x_{\rm f}.
\]
Assume
\[
 x_{\rm f}<(1-\etaP)r_0+\etaP x_{\rm hi},
 \qquad
 x_{\rm f}>(1-\etaM)r_0+\etaM x_{\rm lo}.
\]
Then
\begin{equation}\label{eq:path-reference-gap}
 r_{\rm I}-r_{\rm II}
 =\etaP(1-\etaM)x_{\rm hi}
 -\etaM(1-\etaP)x_{\rm lo}
 +(\etaM-\etaP)x_{\rm f}.
\end{equation}
If the right-hand side is nonzero, the two paths have different regularized CPT values despite identical terminal wealth $x_{\rm f}$.
\end{proposition}

\begin{proposition}[Reference sensitivity]\label{prop:reference-sensitivity}
Under Assumptions \ref{ass:trajectory}--\ref{ass:cpt-regularity}, fix $(P,z,\theta)$.  For this comparison only, suppose the policy features and market transition kernel are reference-isolated, so that changing the episode-start reference from $r$ to $r'$ while using common policy and market random variables leaves the wealth and portfolio paths unchanged.  Set
\begin{equation}\label{eq:reference-constants}
 \eta_{\min}=\min(\etaP,\etaM),
 \qquad
 L_v=\alpha_+\eps_v^{\alpha_+-1}
 +\lambda\alpha_-\eps_v^{\alpha_--1}.
\end{equation}
For two episode-start references $r,r'$ coupled through the same wealth path,
\begin{equation}\label{eq:reference-contraction}
 \abs{r_{t_k+h}(r)-r_{t_k+h}(r')}
 \le(1-\eta_{\min})^h\abs{r-r'},
\end{equation}
and, for fixed $(P,z,\theta)$,
\begin{equation}\label{eq:reference-value-sensitivity}
 \abs{J(P,z,r;\theta)-J(P,z,r';\theta)}
 \le C_1L_v(1-\eta_{\min})^h\abs{r-r'}.
\end{equation}
Consequently, with
\[
 \Delta_k^{\mathrm{nr}}
 :=\sup_{\theta\in\Theta}
 \abs{J(P_{k+1},z_{k+1},r_k;\theta)-J(P_k,z_k,r_k;\theta)},
\]
\begin{equation}\label{eq:variation-decomposition}
 \sup_{\theta\in\Theta}
 \abs{J_{k+1}(\theta,r_{k+1})-J_k(\theta,r_k)}
 \le \Delta_k^{\mathrm{nr}}
 +C_1L_v(1-\eta_{\min})^h\abs{r_{k+1}-r_k}.
\end{equation}
\end{proposition}

\subsection{Exact frozen-target gradient estimation}

\begin{theorem}[Regularized CPT score-gradient identity]\label{thm:gradient-identity}
Under Assumptions \ref{ass:trajectory}--\ref{ass:cpt-regularity}, fix one target and write $J=J_k$, $Y=Y_k$, $S_\theta^\pm(y):=S_{P_k,z_k,r_k,\theta}^\pm(y)$, and $G_\theta:=G_{k,\theta}$.  Define
\begin{align}\label{eq:cpt-multiplier}
 \Psi_\theta(\tau)
 ={}&\int_0^{B_v}(w_+^\eps)'(S_\theta^+(y))
 \{\1[v_+(Y(\tau))>y]-S_\theta^+(y)\}\,dy\notag\\
 &-\int_0^{B_v}(w_-^\eps)'(S_\theta^-(y))
 \{\1[v_-(Y(\tau))>y]-S_\theta^-(y)\}\,dy.
\end{align}
Then
\begin{equation}\label{eq:score-gradient-identity}
 \nabla J(\theta)=\E\{\Psi_\theta(\tau)G_\theta(\tau)\}.
\end{equation}
The same identity holds for the frozen simulator objective $\widetilde J$.
\end{theorem}

The empirical survival probabilities entering \eqref{eq:cpt-multiplier} make a direct finite-sample plug-in gradient biased.  Appendix~\ref{app:estimator} shows that the leave-one-out base has $O(n^{-1})$ mean error and that the coupled full/half difference cancels the first-order influence term, leaving second moment $O(n^{-2}4^{-\ell})$.  These two properties are the mechanism behind the randomized correction in Algorithm~\ref{alg:cloo-rd}.

\begin{theorem}[Exact unbiasedness, finite variance, and finite expected cost]\label{thm:estimator}
Let $Z_n^{\db}$ be defined by \eqref{eq:debiased-output}, with $n\ge2$, $p_{\rm act}\in(0,1]$, and $1<b<2$.  Let $C_U$ and $C_{\rm rem}$ denote the finite constants established by \Cref{lem:level-cancellation}.  Put
\begin{equation}\label{eq:level-series-constants}
 \Xi_b
 :=\sum_{\ell\ge1}\frac{4^{-\ell}}{\nu_\ell}
 =\frac{1}{4(1-2^{-b})(1-2^{b-2})},
 \qquad
 \Upsilon_b
 :=\sum_{\ell\ge1}\nu_\ell2^\ell
 =\frac{2(1-2^{-b})}{1-2^{1-b}}.
\end{equation}
Then
\begin{equation}\label{eq:estimator-unbiased}
 \E Z_n^{\db}=g,
\end{equation}
\begin{equation}\label{eq:single-output-variance}
 \tr\Cov(Z_n^{\db})
 \le C_{\rm var}(n)
 :=\frac{C_U}{n}+\frac{9C_{\rm rem}\Xi_b}{p_{\rm act} n^2}<\infty,
\end{equation}
and the expected number of complete trajectories is
\begin{equation}\label{eq:expected-estimator-cost}
 C_{\rm cost}(n,p_{\rm act},b)
 =n(1+p_{\rm act}\Upsilon_b)<\infty.
\end{equation}
For $M$ independent outputs,
\begin{equation}\label{eq:estimator-mse}
 \widehat g_{M,n}^{\db}:=\frac1M\sum_{m=1}^{M}Z_{n,m}^{\db},
 \qquad
 \E\norm{\widehat g_{M,n}^{\db}-g}_2^2
 \le\frac{C_{\rm var}(n)}{M}.
\end{equation}
\end{theorem}

\begin{theorem}[Fixed-target asymptotic normality]\label{thm:clt}
Under \Cref{thm:estimator}, fix $(n,p_{\rm act},b)$ and let $\Sigma_n=\Cov(Z_n^{\db})$.  Then
\begin{equation}\label{eq:clt}
 \sqrt M\,(\widehat g_{M,n}^{\db}-g)
 \rightsquigarrow \mathcal N(0,\Sigma_n).
\end{equation}
If $\Sigma_n$ is nonsingular and $\widehat\Sigma_{M,n}$ is the empirical covariance of the $M$ outputs, then
\begin{equation}\label{eq:studentized-clt}
 \sqrt M\,\widehat\Sigma_{M,n}^{-1/2}
 (\widehat g_{M,n}^{\db}-g)
 \rightsquigarrow \mathcal N(0,I_d).
\end{equation}
\end{theorem}

The estimator result is therefore not merely an asymptotic bias statement: for each fixed frozen target, the untruncated construction is exactly unbiased while retaining finite variance and finite expected trajectory cost.  The CLT provides replication-based uncertainty quantification for the same estimator.

\subsection{Online first-order tracking under a moving CPT objective}

\begin{theorem}[Average expected squared projected residual bound]\label{thm:online-residual}
Suppose Assumptions \ref{ass:trajectory}--\ref{ass:cpt-regularity} hold and $\gamma L\le\vartheta$.  Let $\widehat g_k^{\db}$ be defined by \eqref{eq:gradient-average} and target $\widetilde g_k(\theta_k)$.  Then
\begin{equation}\label{eq:true-gradient-mse}
 \E\norm{\widehat g_k^{\db}-g_k(\theta_k)}_2^2
 \le\frac{C_{\rm var}(n)}{M_k}+\E\beta_k^2,
\end{equation}
and
\begin{equation}\label{eq:online-residual-bound}
 \boxed{
 \cE_K
 \le
 \frac{8\{2B_v+\E\cV_K\}}{\vartheta\gamma K}
 +\frac{10C_g}{\vartheta^3K}
 \sum_{k=1}^{K}
 \left(\frac{C_{\rm var}(n)}{M_k}+\E\beta_k^2\right),
 }
\end{equation}
\end{theorem}

Equation~\eqref{eq:online-residual-bound} separates the three quantities that govern online tracking: movement of the true objective sequence through $\cV_K$, Monte Carlo error through $M_k^{-1}$, and frozen-simulator discrepancy through $\beta_k^2$.  The target is the contemporaneous constrained first-order condition rather than a single static optimum.

\begin{corollary}[Dynamic local regret]\label{cor:dynamic-local-regret}
For every fixed window length $W\ge1$, discount $\rho\in(0,1]$, and horizon $K\ge1$,
\begin{equation}\label{eq:dynamic-local-regret-bound}
 \Reg_{\rho,W}(K)
 \le \sum_{k=1}^{K}\zeta_k
\end{equation}
pathwise, and hence
\begin{equation}\label{eq:average-dynamic-local-regret-bound}
 \frac1K\E\Reg_{\rho,W}(K)
 \le \cE_K.
\end{equation}
Thus the same projected-residual control also governs the temporally smoothed first-order criterion used in the online evaluation.
\end{corollary}

\paragraph{Stationary-set interpretation.}
The main online guarantee is stated in the projected first-order residual and does not require a metric relation to a particular stationary point.  When parameter-space interpretation is useful, suppose additionally that, for some $\kappa<\infty$,
\begin{equation}\label{eq:error-bound}
 \dist(\theta,\Sta_k)
 \le \kappa\norm{Q_k(\theta)}_2,
 \qquad \theta\in\Theta,
\end{equation}
uniformly over the analyzed episodes.  This local geometric condition is not needed for the residual or DLR bounds; it is used only to translate them into distance from the contemporaneous constrained stationary set.

\begin{corollary}[Moving stationary-set tracking]\label{cor:stationary-tracking}
If \eqref{eq:error-bound} holds, then
\begin{equation}\label{eq:stationary-tracking}
 \frac1K\sum_{k=1}^{K}
 \E\dist^2(\theta_k,\Sta_k)
 \le\kappa^2\cE_K.
\end{equation}
\end{corollary}

\begin{corollary}[Vanishing average expected squared projected residual and DLR]\label{cor:vanishing-tracking}
Fix $W<\infty$ and $\rho\in(0,1]$.  If
\begin{equation}\label{eq:vanishing-conditions}
 \E\cV_K=o(K),
 \qquad
 \frac1K\sum_{k=1}^{K}
 \left(\frac1{M_k}+\E\beta_k^2\right)\to0,
\end{equation}
then
\begin{equation}\label{eq:vanishing-conclusion}
 \cE_K\to0,
 \qquad
 K^{-1}\E\Reg_{\rho,W}(K)\to0.
\end{equation}
If \eqref{eq:error-bound} holds, the left-hand side of \eqref{eq:stationary-tracking} also converges to zero.
\end{corollary}

For the semi-synthetic environment used in E3--E4, Appendix~\ref{app:online} gives a sufficient objective-sensitivity condition under which the diminishing state gain and finite return-law interpolation imply sublinear objective variation.  This makes explicit the bridge between the theorem's variation budget and the experimental construction without building that construction into the general theorem.

\subsection{Fixed-dimension minimax information rate}

\begin{definition}[Complete-trajectory oracle experiment]\label{def:minimax-oracle}
Fix $d$, $\theta_0=0\in\operatorname{int}(\Theta)$, common finite bounds on $h$ and $N$, and the constants in Assumptions \ref{ass:trajectory}--\ref{ass:cpt-regularity}.  Let $\mathfrak C$ be a model class satisfying these bounds, containing the Bernoulli-return submodel constructed in Appendix \ref{app:minimax}, with $(w_+^\eps)'(0)>0$.  Under $P\in\mathfrak C$, an adaptive query $\theta_t\in\Theta$ returns one fresh complete augmented trajectory
\[
 \tau_t\mid\mathcal G_{t-1}\sim P_{\theta_t},
 \qquad
 \mathcal G_t:=\sigma\{(\theta_j,\tau_j):j\le t\}.
\]
An admissible procedure chooses a stopping time $T_{\rm call}$ and an $\mathcal G_{T_{\rm call}}$-measurable estimator $\widehat g$ such that
\begin{equation}\label{eq:minimax-budget}
 \sup_{P\in\mathfrak C}\E_P T_{\rm call}\le\mathsf B,
\end{equation}
with target $g_P:=\nabla J_P(\theta_0)$.
\end{definition}

The corresponding minimax risk is
\begin{equation}\label{eq:minimax-risk}
 \mathfrak R_{\mathsf B}
 :=\inf_{\widehat g:\,\sup_{P\in\mathfrak C}\E_P T_{\rm call}\le\mathsf B}
 \sup_{P\in\mathfrak C}
 \E_P\norm{\widehat g-g_P}_2^2.
\end{equation}

\begin{theorem}[Minimax-optimal fixed-dimension trajectory-budget rate]\label{thm:minimax}
For the oracle experiment in Definition \ref{def:minimax-oracle}, there exist constants
$c_{\rm lb},C_{\rm ub},\mathsf B_0\in(0,\infty)$, independent of $\mathsf B$, such that for all $\mathsf B\ge\mathsf B_0$,
\begin{equation}\label{eq:minimax-rate}
 \frac{c_{\rm lb}}{\mathsf B}
 \le \mathfrak R_{\mathsf B}
 \le \frac{C_{\rm ub}}{\mathsf B}.
\end{equation}
The upper bound is attained, up to constants, by averaging independent untruncated outputs $Z_n^{\db}$ under the same expected complete-trajectory budget.  Hence, for fixed $d$ and fixed regularization, the minimax gradient-estimation rate is $\Theta(\mathsf B^{-1})$.
\end{theorem}
